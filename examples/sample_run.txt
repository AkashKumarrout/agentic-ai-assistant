import math

def factorial_iterative(n):
    """
    Compute the factorial of a number n using an iterative approach.
    
    Args:
    - n (int): The number to compute the factorial of.
    
    Returns:
    - int: The factorial of n.
    """
    if n == 0:
        return 1
    else:
        result = 1
        for i in range(1, n + 1):
            result *= i
        return result

def test_factorial():
    # Test cases with different inputs
    assert factorial_iterative(0) == 1, "Factorial of 0 should be 1"
    assert factorial_iterative(5) == 120, "Factorial of 5 should be 120"
    assert factorial_iterative(1) == 1, "Factorial of 1 should be 1"

# Running the tests
test_factorial()
